cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(dlss5_nr_standalone LANGUAGES CXX)
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "dlss5_nr requires Windows x64")
endif()
if(NOT TARGET CUDA::cuda_driver)
  find_package(CUDAToolkit REQUIRED)
endif()
set(NR_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/nr_embedded")
file(MAKE_DIRECTORY "${NR_GENERATED}")
set(NR_EMBEDDED)
foreach(unit shallow heads2 heads4 heads8 utility deep16 vit bridge)
  set(path "${CMAKE_CURRENT_SOURCE_DIR}/cuda/${unit}.cu")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
  file(READ "${path}" src)
  string(LENGTH "${src}" length)
  set(code "// Generated embedded unchanged CUDA source.\n#include <string>\nnamespace dlss5::nr::detail { std::string source_${unit}() { std::string s; s.reserve(${length});\n")
  set(offset 0)
  while(offset LESS length)
    string(SUBSTRING "${src}" ${offset} 8000 chunk)
    string(APPEND code "s += R\"NRembed(${chunk})NRembed\";\n")
    math(EXPR offset "${offset}+8000")
  endwhile()
  string(APPEND code "return s; } }\n")
  set(out "${NR_GENERATED}/${unit}.cpp")
  file(CONFIGURE OUTPUT "${out}" CONTENT "${code}" @ONLY)
  list(APPEND NR_EMBEDDED "${out}")
endforeach()
add_library(dlss5_nr STATIC nr_engine.cpp ${NR_EMBEDDED})
target_compile_features(dlss5_nr PUBLIC cxx_std_17)
target_include_directories(dlss5_nr PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(dlss5_nr PUBLIC CUDA::cuda_driver)
target_compile_definitions(dlss5_nr PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(dlss5_nr PRIVATE /W4 /EHsc /utf-8)
endif()
option(DLSS5_NR_BUILD_SMOKE "Build the standalone native NR validation executable" OFF)
if(DLSS5_NR_BUILD_SMOKE)
  add_executable(nr_smoke nr_smoke.cpp)
  target_link_libraries(nr_smoke PRIVATE dlss5_nr)
  target_compile_definitions(nr_smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
