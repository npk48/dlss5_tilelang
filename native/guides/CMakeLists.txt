cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
        set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    project(dlss5_guides_standalone LANGUAGES CXX CUDA)
endif()
find_package(CUDAToolkit REQUIRED)
set(DLSS5_ORT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../third_party/ort" CACHE PATH "ONNX Runtime CUDA 1.20+ SDK (CUDA12 / cuDNN9)")
if(NOT EXISTS "${DLSS5_ORT_ROOT}/include/onnxruntime_cxx_api.h")
    message(FATAL_ERROR "Native guides require ONNX Runtime CUDA SDK. See guides/DEPENDENCIES.md; set DLSS5_ORT_ROOT.")
endif()
if(NOT TARGET dlss5_ort)
    add_library(dlss5_ort SHARED IMPORTED GLOBAL)
    set_target_properties(dlss5_ort PROPERTIES
        IMPORTED_IMPLIB "${DLSS5_ORT_ROOT}/lib/onnxruntime.lib"
        IMPORTED_LOCATION "${DLSS5_ORT_ROOT}/runtime/onnxruntime.dll"
        INTERFACE_INCLUDE_DIRECTORIES "${DLSS5_ORT_ROOT}/include")
endif()
add_library(dlss5_guides STATIC guides.cpp kernels.cu)
target_compile_features(dlss5_guides PUBLIC cxx_std_17)
target_include_directories(dlss5_guides PUBLIC "${CMAKE_CURRENT_LIST_DIR}" "${CMAKE_CURRENT_LIST_DIR}/..")
target_link_libraries(dlss5_guides PUBLIC CUDA::cudart_static CUDA::cuda_driver PRIVATE dlss5_ort)
set_target_properties(dlss5_guides PROPERTIES CUDA_ARCHITECTURES 89 CUDA_STANDARD 17 CUDA_RUNTIME_LIBRARY Static POSITION_INDEPENDENT_CODE ON)
target_compile_options(dlss5_guides PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
option(DLSS5_GUIDES_BUILD_VERIFY "Build standalone native/reference fixture runner" ON)
if(DLSS5_GUIDES_BUILD_VERIFY)
    add_executable(guides_verify verify.cpp)
    target_link_libraries(guides_verify PRIVATE dlss5_guides)
    file(GLOB GUIDE_RUNTIME_DLLS "${DLSS5_ORT_ROOT}/runtime/*.dll")
    add_custom_command(TARGET guides_verify POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${GUIDE_RUNTIME_DLLS} "$<TARGET_FILE_DIR:guides_verify>"
        COMMAND_EXPAND_LISTS)
endif()
