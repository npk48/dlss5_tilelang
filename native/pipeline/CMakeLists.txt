cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(dlss5_pipeline LANGUAGES CXX CUDA)
endif()
find_package(CUDAToolkit 12.2 REQUIRED)
add_library(dlss5_pipeline STATIC pipeline.cu)
target_include_directories(dlss5_pipeline PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../include)
target_link_libraries(dlss5_pipeline PUBLIC CUDA::cudart_static CUDA::cuda_driver)
target_compile_features(dlss5_pipeline PUBLIC cxx_std_17)
set_target_properties(dlss5_pipeline PROPERTIES CUDA_STANDARD 17 CUDA_RUNTIME_LIBRARY Static POSITION_INDEPENDENT_CODE ON)
target_compile_options(dlss5_pipeline PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
option(D5_PIPELINE_TEST "Build pipeline numerical test bridge" OFF)
if(D5_PIPELINE_TEST)
  add_library(dlss5_pipeline_test SHARED test_bridge.cpp)
  target_link_libraries(dlss5_pipeline_test PRIVATE dlss5_pipeline)
endif()
