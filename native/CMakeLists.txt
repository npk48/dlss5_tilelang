function(d5_library target kind)
 add_library(${target} ${kind} src/sdk.cpp)
 target_include_directories(${target} PUBLIC include)
 target_link_libraries(${target} PRIVATE dlss5_nr dlss5_pipeline dlss5_reconstruction dlss5_frame_ops CUDA::cuda_driver)
 target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
 if(D5_WITH_GUIDES)
  target_compile_definitions(${target} PRIVATE D5_WITH_GUIDES)
  target_link_libraries(${target} PRIVATE dlss5_guides)
 endif()
 if("${kind}" STREQUAL "STATIC")
  target_compile_definitions(${target} PUBLIC D5_STATIC)
 else()
  set_target_properties(${target} PROPERTIES OUTPUT_NAME dlss5)
 endif()
endfunction()

if(D5_EMBED_DEPENDENCIES)

 endif()

 target_compile_definitions(d5_pack_dependencies PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
 target_link_libraries(d5_pack_dependencies PRIVATE cabinet bcrypt)
 file(GLOB_RECURSE DEP_FILES CONFIGURE_DEPENDS "${DLSS5_ORT_ROOT}/runtime/*.dll" "${D5_TOOLCHAIN}/nvrtc/bin/*" "${D5_TOOLCHAIN}/runtime/include/*" "${D5_TOOLCHAIN}/cccl/include/*")
