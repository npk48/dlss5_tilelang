function(d5_library target kind)
 add_library(${target} ${kind} src/sdk.cpp)
 target_include_directories(${target} PUBLIC include)
 target_link_libraries(${target} PRIVATE dlss5_nr dlss5_pipeline dlss5_reconstruction dlss5_frame_ops CUDA::cuda_driver)
 target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
 if(D5_WITH_GUIDES)
  target_compile_definitions(${target} PRIVATE D5_WITH_GUIDES)
  target_link_libraries(${target} PRIVATE dlss5_guides)
 endif()
 if("${kind}" STREQUAL "STATIC")
  target_compile_definitions(${target} PUBLIC D5_STATIC)
 else()
  set_target_properties(${target} PROPERTIES OUTPUT_NAME dlss5)
 endif()
endfunction()

# The frontend is a real npm build, served from external files. Changes to
# web assets do not require relinking the server executable.
option(D5_BUILD_WEBUI "Build the npm WebUI assets" ON)
if(D5_BUILD_WEBUI)
 find_program(D5_NPM NAMES npm.cmd npm REQUIRED)
 set(D5_WEB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/webui")
 file(GLOB_RECURSE D5_WEB_SOURCES CONFIGURE_DEPENDS "${D5_WEB_DIR}/src/*" "${D5_WEB_DIR}/public/*")
 file(GLOB D5_WEB_CONFIGS CONFIGURE_DEPENDS "${D5_WEB_DIR}/*.json" "${D5_WEB_DIR}/*.ts" "${D5_WEB_DIR}/index.html")
 add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/webui-deps.stamp"
  COMMAND "${D5_NPM}" ci --no-audit --no-fund
  COMMAND ${CMAKE_COMMAND} -E touch "${CMAKE_CURRENT_BINARY_DIR}/webui-deps.stamp"
  WORKING_DIRECTORY "${D5_WEB_DIR}"
  DEPENDS "${D5_WEB_DIR}/package.json" "${D5_WEB_DIR}/package-lock.json" VERBATIM)
 add_custom_command(OUTPUT "${D5_WEB_DIR}/dist/index.html"
  COMMAND "${D5_NPM}" run build
  WORKING_DIRECTORY "${D5_WEB_DIR}"
  DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/webui-deps.stamp" ${D5_WEB_SOURCES} ${D5_WEB_CONFIGS} VERBATIM)
 add_custom_target(dlss5_web_assets
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${D5_WEB_DIR}/dist" "${CMAKE_CURRENT_BINARY_DIR}/assets"
  DEPENDS "${D5_WEB_DIR}/dist/index.html")
 add_dependencies(dlss5_server dlss5_web_assets)
 install(DIRECTORY "${D5_WEB_DIR}/dist/" DESTINATION assets)
endif()
