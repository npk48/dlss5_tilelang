cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(dlss5_reconstruction LANGUAGES CXX CUDA)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(CUDAToolkit 12.2 REQUIRED)
add_library(dlss5_reconstruction STATIC fsr2.cu)
target_include_directories(dlss5_reconstruction PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(dlss5_reconstruction PUBLIC CUDA::cudart_static CUDA::cuda_driver)
target_compile_features(dlss5_reconstruction PUBLIC cxx_std_17)
set_target_properties(dlss5_reconstruction PROPERTIES CUDA_ARCHITECTURES 89 CUDA_STANDARD 17 CUDA_RUNTIME_LIBRARY Static)
target_compile_options(dlss5_reconstruction PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
option(D5_FSR_TEST_BRIDGE "Build Python-independent C ABI validation bridge" OFF)
if(D5_FSR_TEST_BRIDGE)
 target_compile_definitions(dlss5_reconstruction PUBLIC D5_FSR_TEST_API)
 add_library(fsr_test_bridge SHARED ../tools/fsr_test_bridge.cpp)
 target_link_libraries(fsr_test_bridge PRIVATE dlss5_reconstruction)
endif()
